using UnityEngine.Events;

namespace #NAMESPACE#.Runtime
{
    public class BindableProperty<T>
    {
        private T value;
        
        /// <summary>
        /// This Action will be called when the value changes.
        /// </summary>
        public UnityEvent<T> OnChanged = new();

        public T Value
        {
            get { return value; }
            set
            {
                this.value = value;
                OnChanged?.Invoke(this.value);
            }
        }
    }
}
