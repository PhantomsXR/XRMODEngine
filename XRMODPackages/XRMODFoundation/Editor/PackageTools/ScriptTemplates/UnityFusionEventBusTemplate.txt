using System;
using System.Collections.Generic;
using UnityEngine;

namespace #NAMESPACE#.Runtime 
{
    public class UnityFusionEventBus
    {
        private static readonly Dictionary<Type, List<Delegate>> _TYPED_EVENTS = new();
         public static void Subscribe<T>(Action<T> _callback)
        {
            var tmp_Type = typeof(T);
            if (!_TYPED_EVENTS.TryGetValue(tmp_Type, out var tmp_List))
            {
                tmp_List = new List<Delegate>();
                _TYPED_EVENTS[tmp_Type] = tmp_List;
            }

            tmp_List.Add(_callback);
        }

        public static void Unsubscribe<T>(Action<T> _callback)
        {
            var tmp_Type = typeof(T);
            if (!_TYPED_EVENTS.TryGetValue(tmp_Type, out var tmp_List)) return;
            for (int tmp_I = tmp_List.Count - 1; tmp_I >= 0; tmp_I--)
            {
                var tmp_Cb = tmp_List[tmp_I];
                if (tmp_Cb == null || tmp_Cb == (Delegate) _callback)
                {
                    tmp_List.RemoveAt(tmp_I);
                    continue;
                }

                var tmp_CbTypeName = tmp_Cb.GetType().FullName;
                if (tmp_CbTypeName == null ||
                    (!tmp_CbTypeName.Contains("DelegateAdapter") &&
                     !tmp_CbTypeName.Contains("DummyDelegateAdapter")))
                    continue;

                try
                {
                    if (tmp_Cb.Method == _callback.Method && tmp_Cb.Target == _callback.Target)
                    {
                        tmp_List.RemoveAt(tmp_I);
                    }
                }
                catch (Exception tmp_E)
                {
                    Debug.LogError($"[EventBus] Skip DummyDelegateAdapter compare: {tmp_E.Message}");
                }
            }

            if (tmp_List.Count == 0)
                _TYPED_EVENTS.Remove(tmp_Type);
        }

        public static void Publish<T>(T _evt)
        {
            var tmp_Type = typeof(T);
            if (!_TYPED_EVENTS.TryGetValue(tmp_Type, out var tmp_List)) return;

            for (int tmp_Idx = tmp_List.Count - 1; tmp_Idx >= 0; tmp_Idx--)
            {
                var tmp_Del = tmp_List[tmp_Idx];
                if (tmp_Del == null ||
                    tmp_Del.Target is UnityEngine.Object tmp_Obj && !tmp_Obj)
                {
                    tmp_List.RemoveAt(tmp_Idx);
                    continue;
                }

                if (tmp_Del is Action<T> tmp_Action)
                    tmp_Action.Invoke(_evt);
            }
        }

        public static void ClearAll()
        {
            _TYPED_EVENTS.Clear();
        }
    }
}
